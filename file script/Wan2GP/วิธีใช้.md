### เปิดเทอมินอลใหม่ใน wan2gp
ก๊อปไปวางเลย Enter
```
cd /workspace/Wan2GP

# สำรองไฟล์
cp wan/multitalk/multitalk.py wan/multitalk/multitalk.py.backup

# แก้ไขไฟล์
cat > fix_multitalk_v2.py << 'EOF'
# Read the file
with open('wan/multitalk/multitalk.py', 'r') as f:
    lines = f.readlines()

# Find the line with the problematic rearrange
for i, line in enumerate(lines):
    if 'latter_frame_audio_emb = rearrange(latter_frame_audio_emb, "b (n_t n) w s c -> b n_t n w s c", n=vae_scale)' in line:
        # Insert padding code before this line
        indent = '    '  # 4 spaces
        padding_code = [
            indent + '# Fix einops error by padding to make divisible by vae_scale\n',
            indent + 'import torch\n',
            indent + 'current_length = latter_frame_audio_emb.shape[1]\n',
            indent + 'remainder = current_length % vae_scale\n',
            indent + '\n',
            indent + 'if remainder != 0:\n',
            indent + '    padding_needed = vae_scale - remainder\n',
            indent + '    # Pad along the second dimension (frames)\n',
            indent + '    pad_shape = [0, 0, 0, 0, 0, 0, 0, padding_needed]\n',
            indent + '    latter_frame_audio_emb = torch.nn.functional.pad(\n',
            indent + '        latter_frame_audio_emb,\n',
            indent + '        pad_shape,\n',
            indent + '        mode="replicate"\n',
            indent + '    )\n',
            indent + '    print(f"⚠️  Auto-padded audio frames from {current_length} to {current_length + padding_needed} for vae_scale compatibility")\n',
            indent + '\n'
        ]
        
        # Insert the padding code before the rearrange line
        lines[i:i] = padding_code
        break

# Write back to file
with open('wan/multitalk/multitalk.py', 'w') as f:
    f.writelines(lines)

print("✅ Fixed multitalk.py for einops compatibility")
EOF

python fix_multitalk_v2.py
rm fix_multitalk_v2.py
```

### ตามด้วย 
copy วางอีกรอบ
```
cd /workspace/Wan2GP

cat > fix_torch_import.py << 'EOF'
# Read the file
with open('wan/multitalk/multitalk.py', 'r') as f:
    content = f.read()

# Add import torch at the beginning of the function
content = content.replace(
    'def get_window_audio_embeddings(full_audio_embs, audio_start_idx=0, clip_length = 81, vae_scale = 4, audio_window = 5):',
    'def get_window_audio_embeddings(full_audio_embs, audio_start_idx=0, clip_length = 81, vae_scale = 4, audio_window = 5):\n    import torch'
)

# Remove the import torch from inside the if statement
content = content.replace(
    '    if remainder != 0:\n        padding_needed = vae_scale - remainder\n        # Pad along the second dimension (frames)\n        pad_shape = [0, 0, 0, 0, 0, 0, 0, padding_needed]\n        latter_frame_audio_emb = torch.nn.functional.pad(',
    '    if remainder != 0:\n        padding_needed = vae_scale - remainder\n        # Pad along the second dimension (frames)\n        pad_shape = [0, 0, 0, 0, 0, 0, 0, padding_needed]\n        latter_frame_audio_emb = torch.nn.functional.pad('
)

# Remove duplicate import torch if it exists
lines = content.split('\n')
fixed_lines = []
torch_imported = False

for line in lines:
    if 'import torch' in line and 'def get_window_audio_embeddings' in fixed_lines[-1] if fixed_lines else False:
        if not torch_imported:
            fixed_lines.append(line)
            torch_imported = True
    elif 'import torch' in line and 'remainder != 0:' in (fixed_lines[-2] if len(fixed_lines) >= 2 else ''):
        continue  # Skip this import torch
    else:
        fixed_lines.append(line)

content = '\n'.join(fixed_lines)

# Write back
with open('wan/multitalk/multitalk.py', 'w') as f:
    f.write(content)

print("✅ Fixed torch import issue")
EOF

python fix_torch_import.py
rm fix_torch_import.py
```
